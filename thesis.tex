% ── LAMPIRAN ─────────────────────────────────────────────────
% Halaman pembatas "LAMPIRAN" setingkat bab (masuk Daftar Isi). Tiap lampiran
% dibuka dengan \lampiran{judul} → subbab tak bernomor "Lampiran N  judul"
% (pakai \section* + entri TOC manual, sehingga "Lampiran N" mengalir sebagai
% teks dan tidak terpotong kotak nomor TOC). Subbab/gambar/tabel di dalamnya
% diberi prefiks "L<n>." dan direset di tiap lampiran.
\clearpage
\chapter*{LAMPIRAN}
\addcontentsline{toc}{chapter}{LAMPIRAN}

\newcounter{lampno}
\renewcommand{\thelampno}{Lampiran~\arabic{lampno}}      % dipakai oleh \ref → "Lampiran 5"
\renewcommand{\thesubsection}{L\arabic{lampno}.\arabic{subsection}}
\renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}
\renewcommand{\thefigure}{L\arabic{lampno}.\arabic{figure}}
\renewcommand{\thetable}{L\arabic{lampno}.\arabic{table}}
\newcommand{\lampiran}[1]{%
  \clearpage
  \refstepcounter{lampno}%
  \setcounter{subsection}{0}\setcounter{figure}{0}\setcounter{table}{0}%
  \section*{Lampiran~\arabic{lampno}\quad #1}%
  \addcontentsline{toc}{section}{Lampiran~\arabic{lampno}\quad #1}%
}
