% ── Bibliography (biblatex, author-year) ─────────────────────
%  Two-author connector rendered as "dan"; 3+ authors as "et al."
% csquotes has no quote style for Indonesian/bahasai; loaded bare it emits a
% placeholder ("?") around every biblatex title via \mkbibquote. Force the
% English style so titles get proper “ ” quotes. (\DeclareLanguageMapping above
% only remaps biblatex localization strings, NOT csquotes — different package.)
\usepackage[autostyle=false]{csquotes}   % recommended companion for babel + biblatex
\setquotestyle{english}
\usepackage[
  backend     = biber,
  style       = authoryear,
  maxcitenames = 2,
  mincitenames = 1,
  maxbibnames = 99,
  uniquename  = init,    % avoids the giveninits/uniquename=full conflict warning
  giveninits  = true,
  dashed      = false,
  doi         = false,
  isbn        = false,
  url         = false
]{biblatex}
% biblatex ships no Indonesian localization (.lbx); the 'language=english'
% option alone does NOT override babel, so map the document language to
% english.lbx. This resolves every "string untranslated" / "fallback
% definition for \mkbibdate*" warning at once (et al., in, pages, volume,
% edition, month names, ordinals).
\DeclareLanguageMapping{indonesian}{english}
\DeclareLanguageMapping{bahasai}{english}
\addbibresource{config/references.bib}
\renewcommand*{\finalnamedelim}{\addspace dan\addspace}  % two authors -> "dan"
