  \centering
  \adjustbox{max width=\textheight, max totalheight=\textwidth, center}{%
  \begin{tikzpicture}[
    node distance=1.2cm and 1.5cm,
    font=\small,
    basic/.style={draw, text centered, font=\footnotesize,
      minimum height=1.2cm, rounded corners=2mm, text width=4.5cm},
    startend/.style={basic, fill=gray!15, thick, draw=gray!70,
      rounded corners=5mm, text width=3cm},
    process/.style={basic, fill=white, thick, draw=gray!60},
    db/.style={basic, fill=green!5, cylinder, shape border rotate=90,
      aspect=0.15, text width=4.5cm, minimum height=1.5cm, thick,
      draw=green!60!black, rounded corners=0mm},
    io/.style={basic, fill=blue!5, trapezium,
      trapezium left angle=75, trapezium right angle=105,
      thick, draw=blue!70!black, text width=3.8cm, inner sep=2pt},
    condition/.style={basic, fill=yellow!5, diamond, aspect=2,
      thick, draw=orange!80!black, text width=3cm,
      inner sep=0pt, minimum height=1.5cm},
    stagebox/.style={draw, gray!50, thick, dashed,
      rounded corners=3mm, fill=gray!5, inner sep=12pt},
    arrow/.style={thick, -{Latex[length=3mm, width=2mm]},
      draw=gray!80!black},
    elabel/.style={fill=white, font=\scriptsize,
      inner sep=2pt, align=center},
    slabel/.style={font=\bfseries\footnotesize, text=gray!80!black,
      fill=white, inner sep=3pt, yshift=1.5mm}
  ]
    % ==== FASE 1 ====
    \node[startend] (Start) {Mulai};
    \node[process,  below=0.8cm of Start]     (Subj)
      {Pemilihan Domain\\Mata Pelajaran\\(Fisika, Kimia, Biologi)};
    \node[process,  below=0.8cm of Subj]      (CheckCache)
      {Pengecekan Status\\Tembolok Sistem};
    \node[condition,below=0.8cm of CheckCache](IsProcessed)
      {Apakah domain telah\\diproses sebelumnya?};
    \node[startend, right=2cm of IsProcessed] (Terminate)
      {Terminasi /\\Resolusi Pengguna};
    \node[db,       below=1cm of IsProcessed] (ReadKG)
      {Ekstraksi Graf\\Pengetahuan Awal\\(Baseline KG)};
    \node[io,       below=0.8cm of ReadKG]    (ReadExperts)
      {Ingesti Data\\Anotasi Pakar};

    \begin{scope}[on background layer]
      \node[stagebox, fit=(Start)(Terminate)(ReadExperts),
        label={[slabel]above:Fase 1: Inisialisasi Data}] (Fase1) {};
    \end{scope}

    % ==== FASE 2 — CABANG KIRI ====
    \node[db, below left=1.5cm and 0.8cm of ReadExperts] (ContextDB)
      {Pengambilan Konteks\\Buku Teks};
    \node[process, below=0.8cm of ContextDB] (CompareExist)
      {Komparasi Opini Pakar\\(Inter-annotator Agreement)};
    \node[condition, below=0.8cm of CompareExist] (IsAgreeExist)
      {Terdapat konsensus\\pakar?};
    \node[process, below left=1cm and -1.5cm of IsAgreeExist]
      (AcceptExist)
      {Penerimaan Keputusan\\Konsensus};
    \node[process, below right=1cm and -1.8cm of IsAgreeExist]
      (ArbitrateExist)
      {Arbitrase LLM-as-a-Judge\\(Textbook Grounding)};

    % ==== FASE 2 — CABANG KANAN ====
    \node[process, below right=1.5cm and 0.8cm of ReadExperts]
      (CollectNew)
      {Ekstraksi Usulan\\Triple Baru\\(Missing Triples)};
    \node[condition] (IsAgreeNew) at (IsAgreeExist -| CollectNew)
      {Kedua pakar\\menambahkan triple?};
    \node[process, below left=1cm and -1.8cm of IsAgreeNew]
      (ArbitrateNew)
      {Arbitrase Validitas\\Usulan oleh LLM};
    \node[process, below right=1cm and -1.5cm of IsAgreeNew]
      (AcceptNew)
      {Integrasi Triple Baru\\secara Konsensus};

    % ==== FASE 2 — KONVERGENSI ====
    \node[db] (Checkpoint)
      at ($(ArbitrateExist)!0.5!(ArbitrateNew) - (0,2cm)$)
      {Penyimpanan Status\\Arbitrase\\(State Checkpointing)};
    \node[process, below=1.2cm of Checkpoint] (Combine)
      {Agregasi Keputusan Akhir};

    \begin{scope}[on background layer]
      \node[stagebox,
        fit=(ContextDB)(CollectNew)(AcceptExist)(AcceptNew)(Combine),
        label={[slabel]above:Fase 2: Evaluasi \& Arbitrase AI}]
        (Fase2) {};
    \end{scope}

    % ==== FASE 3 ====
    \node[process, below=1.2cm of Combine]    (GoldStandard)
      {Pembentukan\\Gold Standard};
    \node[db,      below=0.8cm of GoldStandard](FinalKG)
      {Konstruksi Final Consensus\\Knowledge Graph};
    \node[startend,below=0.8cm of FinalKG]    (Done) {Selesai};

    \begin{scope}[on background layer]
      \node[stagebox, fit=(GoldStandard)(Done),
        label={[slabel]above:Fase 3: Agregasi \& Penyelesaian}]
        (Fase3) {};
    \end{scope}

    % ==== ARROWS — FASE 1 ====
    \draw[arrow] (Start)       -- (Subj);
    \draw[arrow] (Subj)        -- (CheckCache);
    \draw[arrow] (CheckCache)  -- (IsProcessed);
    \draw[arrow] (IsProcessed) -- node[elabel]{Ya}    (Terminate);
    \draw[arrow] (IsProcessed) -- node[elabel]{Tidak} (ReadKG);
    \draw[arrow] (ReadKG)      -- (ReadExperts);

    % Transisi ke Fase 2
    \draw[arrow] (ReadExperts) -| (ContextDB);
    \draw[arrow] (ReadExperts) -| (CollectNew);

    % ==== ARROWS — FASE 2 KIRI ====
    \draw[arrow] (ContextDB)    -- (CompareExist);
    \draw[arrow] (CompareExist) -- (IsAgreeExist);
    \draw[arrow] (IsAgreeExist) -|
      node[elabel, near start]{Ya}    (AcceptExist);
    \draw[arrow] (IsAgreeExist) -|
      node[elabel, near start]{Tidak} (ArbitrateExist);

    % ==== ARROWS — FASE 2 KANAN ====
    \draw[arrow] (CollectNew)  -- (IsAgreeNew);
    \draw[arrow] (IsAgreeNew)  -|
      node[elabel, near start]{Ya}    (AcceptNew);
    \draw[arrow] (IsAgreeNew)  -|
      node[elabel, near start]{Tidak} (ArbitrateNew);

    % ==== ARROWS — KONVERGENSI ====
    \draw[arrow] (ArbitrateExist) -- (Checkpoint);
    \draw[arrow] (ArbitrateNew)   -- (Checkpoint);
    \draw[arrow] (AcceptExist)    |- (Combine);
    \draw[arrow] (AcceptNew)      |- (Combine);
    \draw[arrow] (Checkpoint)     -- (Combine);

    % ==== ARROWS — FASE 3 ====
    \draw[arrow] (Combine)      -- (GoldStandard);
    \draw[arrow] (GoldStandard) -- (FinalKG);
    \draw[arrow] (FinalKG)      -- (Done);
  \end{tikzpicture}%
  }
