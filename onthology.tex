\section*{1. Visualisasi Struktur Ontologi}

Diagram di bawah ini memvisualisasikan hierarki kelas struktural (Grade $\rightarrow$ Chapter $\rightarrow$ Subtopic $\rightarrow$ Concept) beserta relasi objek (Object Properties) yang mengikatnya.

\begin{figure}[h!]
    \centering
    \begin{tikzpicture}[
        node distance=2cm and 4cm,
        class/.style={rectangle, draw=blue!80!black, fill=blue!5, thick, minimum width=3.5cm, minimum height=1.2cm, rounded corners, align=center, font=\bfseries},
        edge/.style={->, >={Stealth[length=3mm]}, thick, draw=black!70, font=\scriptsize},
        loope/.style={->, >={Stealth[length=3mm]}, thick, draw=green!70!black, font=\scriptsize},
        cross/.style={->, >={Stealth[length=3mm]}, thick, dashed, draw=purple!80!black, font=\scriptsize}
    ]

    % Classes
    \node[class] (grade) {Grade};
    \node[class, below=of grade] (chapter) {Chapter};
    \node[class, below=of chapter] (subtopic) {Subtopic};
    \node[class, below=of subtopic] (concept) {Concept};
    \node[class, right=3cm of concept, fill=gray!10, draw=gray!80] (target) {ConceptTarget\\(Placeholder)};

    % Structural Edges
    \draw[edge] (grade) -- node[right, xshift=2mm] {HAS\_CHAPTER} (chapter);
    \draw[edge] (chapter) -- node[right, xshift=2mm] {HAS\_SUBTOPIC} (subtopic);
    \draw[edge] (subtopic) -- node[right, xshift=2mm] {HAS\_CONCEPT} (concept);

    % Sequential & Pedagogical Edges (Chapter level)
    \path[edge, draw=orange!90!black] (chapter) edge[loop left, out=150, in=210, looseness=4] node[left, align=center] {NEXT\_CHAPTER \\ PRASYARAT \\ MEMPERSIAPKAN} (chapter);

    % Semantic Intra-book (Concept level)
    \path[loope] (concept) edge[loop left, out=150, in=210, looseness=4] node[left, align=right] {BAGIAN\_DARI \\ MENYEBABKAN \\ BERGANTUNG\_PADA \\ MENDEFINISIKAN \\ MEMPENGARUHI \\ BERINTERAKSI\_DENGAN \\ MEMUNGKINKAN \\ MENGATUR} (concept);

    % Semantic Cross-book (Concept level)
    \path[cross] (concept) edge[loop right, out=30, in=-30, looseness=4.5] node[right, align=left] {\textit{LINTAS\_BUKU\_...} \\ SAMA\_DENGAN \\ APLIKASI\_DARI \\ PRASYARAT\_UNTUK \\ MEMPERDALAM \\ BERKAITAN\_DENGAN} (concept);

    \end{tikzpicture}
    \caption{Topologi Ontologi: Relasi Struktural, Sekuensial, dan Semantik.}
    \label{fig:ontology}
\end{figure}

\section*{2. Spesifikasi Kelas dan Properti}

\subsection*{2.1 Definisi Kelas (Classes)}
\begin{table}[h!]
    \centering
    \renewcommand{\arraystretch}{1.3}
    \begin{tabularx}{\textwidth}{@{}l X@{}}
        \toprule
        \textbf{Kelas} & \textbf{Deskripsi} \\
        \midrule
        \texttt{Grade} & Tingkat kelas pada Kurikulum Merdeka (mis. Kelas XI atau XII). Akar hierarki. \\
        \texttt{Chapter} & Bab dalam buku teks, bersumber dari Daftar Isi. Memiliki ringkasan akademik. \\
        \texttt{Subtopic} & Subbab di bawah Chapter. Mengelompokkan Concept dalam satu unit. \\
        \texttt{Concept} & Konsep ilmiah yang diekstrak dari teks. Bersifat \textit{shared} lintas dokumen. \\
        \texttt{ConceptTarget} & \textit{Forward reference}: konsep target yang eksis sementara sebelum diekstrak penuh. \\
        \bottomrule
    \end{tabularx}
\end{table}
